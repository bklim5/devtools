    ```bash
    assert_no_keygen_strings() {
      local app="${1:?usage: assert_no_keygen_strings <path-to-signed.app>}"
      local resources="$app/Contents/Resources"
      if [[ ! -d "$resources" ]]; then
        echo "FAIL: bundle Resources dir not found at '$resources'" >&2
        return 1
      fi
      # Forbidden Keygen literals (MAS-BUILD-04, D-03) — any hit means the Keygen
      # surface survived tree-shaking into the store bundle (a stray non-static import).
      local patterns=( 'license\.tinkerdev\.io' 'tinkerdev\.io/buy' 'I have a license key' )
      local hit=0
      for pat in "${patterns[@]}"; do
        if grep -rIlE "$pat" "$resources" >/dev/null 2>&1; then
          echo "FAIL: forbidden Keygen string '$pat' present in the store bundle (Keygen surface did not tree-shake out)" >&2
          grep -rIlE "$pat" "$resources" | sed 's/^/    /' >&2
          hit=1
        fi
      done
      [[ "$hit" -eq 0 ]] || return 1
      echo "OK: no Keygen strings (license.tinkerdev.io / buy link / key-field markers) in the store bundle"
    }
    ```
